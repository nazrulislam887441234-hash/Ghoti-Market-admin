নিরাপত্তা নির্দেশনা: Firestore Security Rules

Owner ও Active Admin অ্যাক্সেস কন্ট্রোল রুলস

আপনার Firebase Console -> Firestore Database -> Rules সেকশনে নিচের রুলস নিশ্চিত করুন:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper function for Owner check
    function isOwner() {
      return request.auth != null && 
        (request.auth.token.email.lower() == 'nazrulislam887441234@gmail.com' || 
         request.auth.token.email.lower() == 'support@ghotimarket.com');
    }
    
    // Helper function for Admin check
    function isActiveAdmin() {
      return request.auth != null && 
        exists(/databases/$(database)/documents/admins/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.uid == request.auth.uid &&
        get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.active == true;
    }

    // Rules for ImgBB API collection
    match /imgbb_api/{documentId} {
      allow read, create, update, delete: if isOwner() || isActiveAdmin();
    }
    
    // Admins collection rule
    match /admins/{adminId} {
      allow read: if request.auth != null;
    }
  }
}
