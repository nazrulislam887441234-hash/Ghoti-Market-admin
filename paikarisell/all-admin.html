rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isOwner() {
      return request.auth != null && 
        (request.auth.token.email.toLowerCase() == "nazrulislam887441234@gmail.com" || 
         request.auth.token.email.toLowerCase() == "support@ghotimarket.com");
    }
    match /admins/{adminUid} {
      allow read, create, update, delete: if isOwner();
      allow read: if request.auth != null && request.auth.uid == adminUid;
    }
  }
}
