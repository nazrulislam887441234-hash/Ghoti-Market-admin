Firestore সিকিউরিটি রুলস কনফিগারেশন

অনুমোদিত ওনার ও অ্যাডমিন সিকিউরিটি নিশ্চিত করতে এই রুলসগুলো ফায়ারবেসে যুক্ত করুন

firestore.rules 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper functions
    function isOwner() {
      return request.auth != null && 
        (request.auth.token.email.trim().toLowerCase() == 'nazrulislam887441234@gmail.com' ||
         request.auth.token.email.trim().toLowerCase() == 'support@ghotimarket.com');
    }

    function isActiveAdmin() {
      return request.auth != null && 
        exists(/databases/$(database)/documents/admins/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.uid == request.auth.uid &&
        get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.active == true;
    }

    function isOwnerOrAdmin() {
      return isOwner() || isActiveAdmin();
    }

    // Admins Collection
    match /admins/{adminUid} {
      allow read: if request.auth != null;
      allow write: if isOwner();
    }

    // Suppliers Collection
    match /supplyers/{supplierUid} {
      allow read: if request.auth != null;
      allow write: if isOwnerOrAdmin();
    }

    // Transactions Collection
    match /transactions/{transactionId} {
      allow read: if isOwnerOrAdmin() || (request.auth != null && resource.data.uid == request.auth.uid);
      allow create: if isOwnerOrAdmin();
      allow update, delete: if isOwnerOrAdmin();
    }
  }
}
