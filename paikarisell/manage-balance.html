service cloud.firestore {
  match /databases/{database}/documents {
    function isOwner() {
      return request.auth != null && 
        (request.auth.token.email.toLowerCase() == "nazrulislam887441234@gmail.com" || 
         request.auth.token.email.toLowerCase() == "support@ghotimarket.com");
    }
    function isAdmin() {
      return request.auth != null && 
        exists(/databases/$(database)/documents/admins/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.active == true;
    }
    match /supplyers/{uid} {
      allow read, update: if isOwner() || isAdmin();
    }
    match /transactions/{txId} {
      allow create, read: if isOwner() || isAdmin();
    }
  }
}
